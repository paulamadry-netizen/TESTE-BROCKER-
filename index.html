// ========================= // MINI-BROKER — JS PRO // ========================= // ========================= // Variables globales // ========================= let currentSymbol = "AAPL"; let currentPrice = 0; let orders = []; // ordres ouverts let orderHistory = []; // ordres fermés let orderIdCounter = 1; let tvFeed = null; // ========================= // Initialisation TradingView // ========================= function initTradingView() { const widget = new TradingView.widget({ symbol: currentSymbol, interval: "1", container_id: "chart", datafeed: new Datafeeds.UDFCompatibleDatafeed( "https://demo_feed.tradingview.com" ), library_path: "https://s3.tradingview.com/", locale: "fr", theme: "dark", autosize: true }); setTimeout(() => attachRealtimeFeed(), 2000); } // ========================= // Flux temps réel TradingView (simulation) // ========================= function attachRealtimeFeed() { if (tvFeed) clearInterval(tvFeed); tvFeed = setInterval(() => { // Simulation du prix (à remplacer si tu as un vrai flux) const randomMove = (Math.random() - 0.5) * 0.2; currentPrice = Math.max(0, (currentPrice || 150) + randomMove); updatePriceUI(); processActiveOrders(); }, 500); } // ========================= // Mise à jour affichage prix // ========================= function updatePriceUI() { document.getElementById("currentPrice").textContent = currentPrice.toFixed(2); document.getElementById("priceInput").placeholder = currentPrice.toFixed(2); } // ========================= // Gestion des tabs // ========================= function activateTab(tab) { document.querySelectorAll(".tab").forEach(t => t.classList.remove("active")); tab.classList.add("active"); const target = tab.dataset.tab; document.getElementById("broker").style.display = target === "trade" ? "block" : "none"; document.getElementById("ordersTab").style.display = target === "orders" ? "block" : "none"; document.getElementById("historyTab").style.display = target === "history" ? "block" : "none"; } // ========================= // Création d'un ordre // ========================= function createOrder(type, side, price, lots, sl, tp) { return { id: orderIdCounter++, time: new Date().toLocaleTimeString(), type, side, lots: parseFloat(lots), entry: price, sl: sl ? parseFloat(sl) : null, tp: tp ? parseFloat(tp) : null, status: "open" }; } // ========================= // TRAITEMENT — Ordre MARCHÉ // ========================= function executeMarketOrder(side) { const lots = parseFloat(document.getElementById("lotInput").value); const sl = document.getElementById("slInput").value || null; const tp = document.getElementById("tpInput").value || null; if (!lots || lots <= 0) { alert("Lots invalides"); return; } const order = createOrder("market", side, currentPrice, lots, sl, tp); orders.push(order); refreshOrdersTable(); } // ========================= // TRAITEMENT — Ordre LIMITE // ========================= function executeLimitOrder(side) { const price = parseFloat(document.getElementById("priceInput").value); const lots = document.getElementById("lotInput").value; const sl = document.getElementById("slInput").value || null; const tp = document.getElementById("tpInput").value || null; if (!price || price <= 0) { alert("Prix limite invalide"); return; } if (!lots || lots <= 0) { alert("Lots invalides"); return; } const order = createOrder("limit", side, price, lots, sl, tp); orders.push(order); refreshOrdersTable(); } // ========================= // Vérifie les ordres ouverts // ========================= function processActiveOrders() { for (let i = orders.length - 1; i >= 0; i--) { const o = orders[i]; // Ordre LIMIT → activation if (o.type === "limit" && o.status === "open") { if ( (o.side === "BUY" && currentPrice <= o.entry) || (o.side === "SELL" && currentPrice >= o.entry) ) { o.entry = currentPrice; o.status = "running"; } } // Ordres en cours (SL/TP) if (o.status === "running") { if (o.tp && ((o.side === "BUY" && currentPrice >= o.tp) || (o.side === "SELL" && currentPrice <= o.tp))) { closeOrder(i); } if (o.sl && ((o.side === "BUY" && currentPrice <= o.sl) || (o.side === "SELL" && currentPrice >= o.sl))) { closeOrder(i); } } } refreshOrdersTable(); } // ========================= // Clôture d’un ordre // ========================= function closeOrder(i) { const o = orders[i]; o.status = "closed"; o.exit = currentPrice; o.pnl = (o.side === "BUY" ? (currentPrice - o.entry) : (o.entry - currentPrice)) * o.lots; orderHistory.push(o); orders.splice(i, 1); refreshOrdersTable(); refreshHistoryTable(); } // ========================= // Table des ordres ouverts // ========================= function refreshOrdersTable() { const tbody = document.querySelector("#ordersTable tbody"); tbody.innerHTML = ""; orders.forEach(o => { const tr = document.createElement("tr"); tr.className = o.side === "BUY" ? "buy-row" : "sell-row"; tr.innerHTML = ` ${o.id} ${o.time} ${o.type} ${o.side} ${o.entry.toFixed(2)} ${o.sl ?? "-"} ${o.tp ?? "-"} ${o.lots} ${o.status} `; tbody.appendChild(tr); }); } // ========================= // Table historique // ========================= function refreshHistoryTable() { const tbody = document.querySelector("#historyTable tbody"); tbody.innerHTML = ""; orderHistory.forEach(o => { const tr = document.createElement("tr"); tr.className = o.pnl >= 0 ? "buy-row" : "sell-row"; tr.innerHTML = ` ${o.id} ${o.type} ${o.side} ${o.entry.toFixed(2)} ${o.exit.toFixed(2)} ${o.lots} ${o.pnl.toFixed(2)} ${o.time} `; tbody.appendChild(tr); }); } // ========================= // Bouton FERMER (manuel) // ========================= function forceClose(id) { const i = orders.findIndex(o => o.id === id); if (i !== -1) closeOrder(i); } // ========================= // Changement de symbole // ========================= function setSymbolFromInput() { const input = document.getElementById("symbolInput"); const value = input.value.trim().toUpperCase(); if (!value) return; currentSymbol = value; document.getElementById("symbolDisplay").textContent = currentSymbol; initTradingView(); } // ========================= // ÉVÉNEMENTS // ========================= document.getElementById("btnBuy").onclick = () => executeMarketOrder("BUY"); document.getElementById("btnSell").onclick = () => executeMarketOrder("SELL"); document.getElementById("btnBuyLimit").onclick = () => executeLimitOrder("BUY"); document.getElementById("btnSellLimit").onclick = () => executeLimitOrder("SELL"); document.querySelectorAll(".tab").forEach(tab => tab.addEventListener("click", () => activateTab(tab)) ); document.getElementById("symbolInput").addEventListener("keydown", e => { if (e.key === "Enter") setSymbolFromInput(); }); // ========================= // Démarrage // ========================= window.onload = () => { document.getElementById("symbolDisplay").textContent = currentSymbol; initTradingView(); };
